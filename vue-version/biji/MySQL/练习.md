```
);

-- 5. JSON 用户信息表
CREATE TABLE user_profile (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_name VARCHAR(50) NOT NULL,
    ext_info JSON
);
```

#### 题目 6：性能优化题（慢查询优化）

sql











```sql
SELECT 
  u.name, o.order_time, p.category
FROM users u
LEFT JOIN orders o ON u.id = o.user_id
LEFT JOIN order_items oi ON o.id = oi.order_id
LEFT JOIN products p ON oi.product_id = p.id

  u.register_time > '2023-01-01'

```

### 题目 2：窗口函数 + 排名（用户消费分层）

sql











```sql
WITH user_consume AS (
  -- 统计每个用户2024年总消费
  SELECT
    u.id AS user_id,
    u.city,
    SUM(o.total_amount) AS total_consume,
    -- 消费分层
    CASE
      WHEN SUM(o.total_amount) >= 10000 THEN 'VIP用户'
      WHEN SUM(o.total_amount) >= 5000 THEN '高级用户'
      WHEN SUM(o.total_amount) >= 1000 THEN '普通用户'
      ELSE '新用户'
    END AS user_level
  FROM users u
  LEFT JOIN orders o ON u.id = o.user_id AND YEAR(o.order_time) = 2024 AND o.status = 1
  GROUP BY u.id, u.city
)
SELECT
  user_id,

  total_consume,
  user_level,

  DENSE_RANK() OVER(PARTITION BY city, user_level ORDER BY total_consume DESC) AS rank_in_level
FROM user_consume;
```

#### 2. 思路解析

- `PARTITION BY city, user_level`按 “城市 + 分层” 分组排名。

```sql
WITH RECURSIVE emp_hierarchy AS (
  -- 递归起点：顶级管理者（manager_id=0）
  SELECT
    id,
    name,
    manager_id,
    salary,
    department_id,
    CAST(name AS CHAR(200)) AS path -- 层级路径
  FROM employee WHERE manager_id = 0
  UNION ALL
  -- 递归体：关联下级员工
  SELECT
    e.id,
    e.name,
    e.manager_id,
    e.salary,
    e.department_id,
    CONCAT(eh.path, '→', e.name) AS path
  FROM employee e
  JOIN emp_hierarchy eh ON e.manager_id = eh.id
)
SELECT
  eh.name,
  eh.salary,
  m.name AS manager_name,
  eh.path,
  -- 同部门同层级链薪资排名
  RANK() OVER(PARTITION BY eh.department_id, eh.path ORDER BY eh.salary DESC) AS salary_rank
FROM emp_hierarchy eh
```

```sql
SELECT
  nfl.first_login_date,
  COUNT(DISTINCT nfl.user_id) AS total_new_user, -- 首日登录用户数
  -- 次日留存率
  ROUND(COUNT(DISTINCT CASE WHEN DATEDIFF(ul.login_date, nfl.first_login_date) = 1 THEN nfl.user_id ELSE NULL END) / COUNT(DISTINCT nfl.user_id), 4) AS next_day_retention,
```

### 题目 5：复杂子查询 + 数据去重（订单去重统计）

#### 1. 建表 & 测试数据

sql











```sql
CREATE TABLE orders_dup (
  id INT PRIMARY KEY,
  user_id INT,
  product_id INT,
  order_time DATETIME,
  total_amount DECIMAL(10,2)

INSERT INTO orders_dup VALUES
(1, 1, 1, '2024-01-01 10:00:00', 100.00),
(2, 1, 1, '2024-01-01 10:00:00', 100.00), -- 重复订单
(3, 2, 2, '2024-01-10 11:00:00', 200.00),
(4, 1, 2, '2024-02-01 12:00:00', 150.00);
```

#### 2. 核心 SQL

sql











```sql
-- 步骤1：去重（保留order_id最小的记录）
WITH order_deduplicate AS (
  SELECT
    *,
    MIN(id) OVER(PARTITION BY user_id, product_id, order_time) AS min_order_id
  FROM orders_dup
),
valid_orders AS (
  SELECT * FROM order_deduplicate WHERE id = min_order_id
),
-- 步骤2：统计重复订单数
dup_stats AS (
  SELECT
    QUARTER(order_time) AS quarter,
    COUNT(*) AS total_order,
    COUNT(*) - COUNT(DISTINCT CONCAT(user_id, product_id, order_time)) AS dup_order
  FROM orders_dup
  GROUP BY QUARTER(order_time)
)
-- 最终统计
SELECT
  -- 每月去重后订单数
  DATE_FORMAT(vo.order_time, '%Y-%m') AS month,
  COUNT(vo.id) AS dedup_order_count,
  -- 每个商品的月均下单用户数
  AVG(monthly_user_count) AS avg_monthly_user_per_product,
  -- 季度重复订单比例
  ds.dup_order / ds.total_order AS dup_ratio
FROM valid_orders vo
LEFT JOIN (
  SELECT product_id, DATE_FORMAT(order_time, '%Y-%m') AS month, COUNT(DISTINCT user_id) AS monthly_user_count
  FROM valid_orders
  GROUP BY product_id, month
) pu ON vo.product_id = pu.product_id AND DATE_FORMAT(vo.order_time, '%Y-%m') = pu.month
LEFT JOIN dup_stats ds ON QUARTER(vo.order_time) = ds.quarter
GROUP BY month, ds.dup_ratio;
```

#### 3. 思路解析

- `MIN(id) OVER(PARTITION BY...)`按 “用户 + 商品 + 时间” 分组，取最小 ID 作为有效订单；
- `CONCAT(user_id, product_id, order_time)`拼接字段，统计重复订单数；
- 子查询`pu`先统计每个商品每月下单用户数，再计算月均。

### 题目 6：性能优化题（慢查询优化）

#### 1. 慢查询原因分析

| 问题点          | 具体说明                                                     |
| --------------- | ------------------------------------------------------------ |
| 过滤顺序错误    | `LEFT JOIN`后才过滤`o.status=1`和`p.category='电子产品'`，导致无效数据参与 JOIN，数据量变大 |
| 缺少索引        | `users(register_time)`、`orders(user_id, status, order_time)`、`order_items(order_id, product_id)`、`products(id, category)`无索引，全表扫描 |
| JOIN 顺序不合理 | MySQL 优化器可能选择低效的 JOIN 顺序（如先扫大表）           |

#### 2. 优化方案

##### （1）索引设计

sql











```sql
-- 用户表：按注册时间筛选，加索引
CREATE INDEX idx_users_register_time ON users(register_time);
-- 订单表：user_id+status+order_time联合索引（覆盖WHERE+JOIN+ORDER BY）
CREATE INDEX idx_orders_user_status_time ON orders(user_id, status, order_time);
-- 订单项表：order_id+product_id联合索引（覆盖JOIN）
CREATE INDEX idx_order_items_order_product ON order_items(order_id, product_id);
-- 商品表：id+category联合索引（覆盖JOIN+WHERE）
CREATE INDEX idx_products_id_category ON products(id, category);
```

##### （2）SQL 改写（先过滤再 JOIN）

sql











```sql
SELECT 
  u.name, o.order_time, p.category
FROM users u
JOIN orders o ON u.id = o.user_id
JOIN order_items oi ON o.id = oi.order_id
JOIN products p ON oi.product_id = p.id
WHERE 
  u.register_time > '2023-01-01' 
  AND o.status = 1 
  AND p.category = '电子产品'
ORDER BY o.order_time DESC;
```

- 把`LEFT JOIN`改为`JOIN`（因 WHERE 条件过滤了非电子产品 / 未支付订单，LEFT JOIN 无意义）；
- 先过滤小表（如 products），再 JOIN 大表，减少参与 JOIN 的数据量。

### 题目 8：事务 + 锁（高并发场景）

#### 1. 建表 & 测试数据

sql











```sql
-- 商品库存表
CREATE TABLE product_stock (
  product_id INT PRIMARY KEY,
  stock INT COMMENT '库存数量'
);
-- 订单表
CREATE TABLE seckill_orders (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user_id INT,
  product_id INT,
  create_time DATETIME
);
-- 订单日志表
CREATE TABLE order_log (
  id INT PRIMARY KEY AUTO_INCREMENT,
  user_id INT,
  product_id INT,
  order_id INT,
  status VARCHAR(10) COMMENT '成功/失败',
  msg VARCHAR(100),
  create_time DATETIME
);
INSERT INTO product_stock VALUES (1, 10); -- 商品1库存10
```

#### 2. 核心 SQL（秒杀下单事务）

sql











```sql
-- 开启事务
START TRANSACTION;
SET @user_id = 1; -- 下单用户ID
SET @product_id = 1; -- 秒杀商品ID

-- 步骤1：加行锁查询库存（防止超卖）
SELECT stock INTO @stock FROM product_stock WHERE product_id = @product_id FOR UPDATE;

-- 步骤2：判断库存是否充足
IF @stock > 0 THEN
  -- 步骤3：扣减库存
  UPDATE product_stock SET stock = stock - 1 WHERE product_id = @product_id;
  -- 步骤4：创建订单
  INSERT INTO seckill_orders (user_id, product_id, create_time) VALUES (@user_id, @product_id, NOW());
  SET @order_id = LAST_INSERT_ID();
  -- 步骤5：记录成功日志
  INSERT INTO order_log (user_id, product_id, order_id, status, msg, create_time) 
  VALUES (@user_id, @product_id, @order_id, '成功', '秒杀下单成功', NOW());
  COMMIT; -- 提交事务
  SELECT '下单成功' AS result;
ELSE
  -- 步骤6：库存不足，记录失败日志
  INSERT INTO order_log (user_id, product_id, order_id, status, msg, create_time) 
  VALUES (@user_id, @product_id, NULL, '失败', '库存不足', NOW());
  ROLLBACK; -- 回滚事务
  SELECT '库存不足，下单失败' AS result;
END IF;
```

#### 3. 思路解析

- `SELECT ... FOR UPDATE`加行锁，防止高并发下多个事务同时读取库存；
- 事务包裹 “扣库存 + 创订单 + 记日志”，保证原子性；
- `LAST_INSERT_ID()`获取新增订单 ID，便于日志关联；
- 库存不足时回滚事务，仅记录失败日志。

#### 2. 核心 SQL

```sql
SELECT
  -- 1. 每个省份的用户数
  JSON_UNQUOTE(JSON_EXTRACT(up.ext_info, '$.address.province')) AS province,
  COUNT(DISTINCT up.user_id) AS province_user_count,
  -- 2. 同时含VIP和高消费标签的用户数
  SUM(CASE 
    WHEN JSON_CONTAINS(up.ext_info->'$.tags', '"VIP"')

    THEN 1 ELSE 0 
  END) AS vip_high_consume_count,

    AND up.user_id NOT IN (
      SELECT DISTINCT o.user_id FROM orders o
      JOIN order_items oi ON o.id = oi.order_id
      JOIN products p ON oi.product_id = p.id
      WHERE p.category = '电子产品'
    ) THEN 1 ELSE 0 
  END) AS electronic_prefer_no_order_count
FROM user_profile up
GROUP BY province;
```

#### 3. 思路解析

- 子查询筛选下单过电子产品的用户，用`NOT IN`对比偏好用户。

## 题目 1：高并发库存扣减（防超卖 + 防重复下单）

### 场景 & 需求回顾

电商秒杀场景，实现：① 同一用户同商品仅下单 1 次；② 乐观锁扣库存防超卖；③ 事务原子性；④ 高并发兼容。

### 1. 建表语句

sql











```sql
-- 商品库存表（乐观锁版本号）
```

### 2. 核心事务 SQL（乐观锁方案）

sql











```sql
-- 设置事务隔离级别（读已提交，减少锁等待）
SET TRANSACTION ISOLATION LEVEL READ COMMITTED;
START TRANSACTION;

-- 定义变量
SET @user_id = 1001;
SET @product_id = 1;

-- 步骤1：检查是否重复下单（利用唯一索引快速判断）
INSERT INTO user_seckill (user_id, product_id, status, create_time)
VALUES (@user_id, @product_id, 0, NOW())
ON DUPLICATE KEY UPDATE 
  status = 0; -- 重复则不修改，仅触发唯一索引冲突

-- 步骤2：乐观锁扣减库存（版本号匹配才更新）
UPDATE product_stock 
SET stock = stock - 1, version = version + 1
WHERE product_id = @product_id 
  AND stock > 0 
  AND version = (SELECT version FROM product_stock WHERE product_id = @product_id);

-- 步骤3：判断库存扣减是否成功
SET @update_rows = ROW_COUNT(); -- 获取更新行数
IF @update_rows > 0 THEN
  -- 扣库存成功：更新秒杀记录状态
  UPDATE user_seckill 
  SET status = 1 
  WHERE user_id = @user_id AND product_id = @product_id;
  
  -- 记录成功日志
  INSERT INTO seckill_log (user_id, product_id, operate_type, result, msg)
  VALUES (@user_id, @product_id, '扣库存', '成功', CONCAT('用户', @user_id, '秒杀商品', @product_id, '成功，剩余库存:', (SELECT stock FROM product_stock WHERE product_id = @product_id)));
  
  COMMIT;
  SELECT '秒杀成功' AS result;
ELSE
  -- 扣库存失败：回滚事务
  ROLLBACK;
  
  -- 记录失败日志
  INSERT INTO seckill_log (user_id, product_id, operate_type, result, msg)
  VALUES (@user_id, @product_id, '扣库存', '失败', IF((SELECT COUNT(*) FROM user_seckill WHERE user_id = @user_id AND product_id = @product_id) > 1, '重复下单', '库存不足或版本冲突'));
  
  SELECT '秒杀失败' AS result;
END IF;
```

### 3. 关键解析

- **防重复下单**：`uk_user_product` 唯一索引，避免同一用户重复插入；
- **乐观锁防超卖**：通过`version`版本号保证 “读 - 改 - 写” 原子性，高并发下无锁等待（QPS 可达 1000+）；
- **事务隔离级别**：读已提交（RC）比可重复读（RR）锁粒度更细，减少死锁；
- **性能优化**：所有查询均基于主键 / 唯一索引，无全表扫描。

```sql
  SELECT user_id 
  FROM user_behavior 
  WHERE behavior_time BETWEEN '2024-07-01' AND '2024-09-30'
  GROUP BY user_id, HOUR(behavior_time)
  HAVING COUNT(*) > 100
),
-- 步骤2：统计各行为用户数（按渠道）

  SELECT
    uc.channel,
    COUNT(DISTINCT CASE WHEN ub.behavior_type = '浏览' THEN ub.user_id END) AS view_user,

  FROM user_behavior ub
  LEFT JOIN user_channel uc ON ub.user_id = uc.user_id
  WHERE ub.behavior_time BETWEEN '2024-07-01' AND '2024-09-30'
    AND ub.user_id NOT IN (SELECT user_id FROM abnormal_user)
  GROUP BY uc.channel
),
-- 步骤3：用户行为路径
user_path AS (
  SELECT
    user_id,
    GROUP_CONCAT(page ORDER BY behavior_time SEPARATOR '→') AS path
  FROM user_behavior
  WHERE user_id NOT IN (SELECT user_id FROM abnormal_user)
  GROUP BY user_id
),
-- 步骤4：加购未下单TOP商品
cart_no_order AS (
  SELECT
    product_id,
    COUNT(DISTINCT ub.user_id) AS user_count
  FROM user_behavior ub
  WHERE ub.behavior_type = '加购'
    AND ub.user_id NOT IN (
      SELECT DISTINCT user_id FROM user_behavior WHERE behavior_type = '下单' AND product_id = ub.product_id
    )
    AND ub.user_id NOT IN (SELECT user_id FROM abnormal_user)
  GROUP BY product_id
  ORDER BY user_count DESC
  LIMIT 10
)

-- 最终输出：漏斗转化率+行为路径+TOP商品
SELECT
  fd.channel,
  fd.view_user,
  fd.cart_user,
  ROUND(fd.cart_user / fd.view_user, 4) AS view2cart, -- 浏览→加购转化率
  ROUND(fd.order_user / fd.cart_user, 4) AS cart2order, -- 加购→下单转化率
  ROUND(fd.pay_user / fd.order_user, 4) AS order2pay, -- 下单→支付转化率
  up.path AS sample_user_path,
  cno.product_id AS top_cart_no_order_product,
  cno.user_count AS top_user_count
FROM funnel_data fd
CROSS JOIN user_path up LIMIT 1 -- 取1条行为路径示例
CROSS JOIN cart_no_order cno LIMIT 1; -- 取1条TOP商品示例
```

### 3. 关键解析

- **刷单用户过滤**：按 “用户 + 小时” 分组统计行为次数，排除异常用户；
- **漏斗转化率**：`CASE WHEN + COUNT(DISTINCT)` 统计各环节用户数，计算转化率；
- **行为路径**：`GROUP_CONCAT(page ORDER BY behavior_time)` 按时间拼接路径；
- **性能优化**：CTE 分层拆分逻辑，避免多层嵌套子查询；优先过滤异常用户，减少后续计算量。

## 题目 3：慢 SQL 优化（千万级订单表查询优化）

### 场景 & 需求回顾

千万级`orders`表慢查询优化，分析根因 + 优化方案 + 归档策略。

### 1. 原慢查询 & 根因分析

sql











```sql
-- 原慢查询
SELECT 
  DATE(pay_time) AS pay_date,
  COUNT(DISTINCT user_id) AS pay_user,
  SUM(total_amount) AS pay_amount
FROM orders
WHERE 
  status = 2 
  AND pay_time BETWEEN '2024-01-01' AND '2024-12-31'
  AND user_id IN (SELECT user_id FROM user_tag WHERE tag = '高价值')
GROUP BY pay_date
ORDER BY pay_amount DESC;
```

#### 慢查询根因

| 问题点     | 具体说明                                                     |
| ---------- | ------------------------------------------------------------ |
| 索引缺失   | `orders`无`status+pay_time+user_id`联合索引，全表扫描；`user_tag`无`tag+user_id`索引，子查询慢 |
| 子查询低效 | `IN`子查询在千万级数据下性能差，改为`JOIN`更高效             |
| 聚合成本高 | `COUNT(DISTINCT user_id)` 对千万级数据聚合耗时久             |
| 数据量过大 | 全表扫描 2024 年全年数据，无分区表隔离历史数据               |

### 2. 优化方案

#### （1）索引设计（核心）

sql











```sql
-- orders表：覆盖WHERE+GROUP BY+ORDER BY的联合索引
CREATE INDEX idx_orders_status_paytime_user ON orders(status, pay_time, user_id, total_amount);

-- user_tag表：标签+用户ID索引
CREATE INDEX idx_user_tag_tag_user ON user_tag(tag, user_id);
```

#### （2）SQL 改写（JOIN 替代子查询 + 先过滤后聚合）

sql











```sql
-- 优化后SQL
WITH high_value_user AS (
  SELECT DISTINCT user_id FROM user_tag WHERE tag = '高价值'
)
SELECT 
  DATE(o.pay_time) AS pay_date,
  COUNT(DISTINCT o.user_id) AS pay_user,
  SUM(o.total_amount) AS pay_amount
FROM orders o
JOIN high_value_user hv ON o.user_id = hv.user_id
WHERE 
  o.status = 2 
  AND o.pay_time BETWEEN '2024-01-01' AND '2024-12-31'
GROUP BY pay_date
ORDER BY pay_amount DESC;
```

#### （3）分表 / 分区策略（千万级数据）

sql











```sql
-- 按pay_time创建分区表（按年分区）
ALTER TABLE orders 
PARTITION BY RANGE (TO_DAYS(pay_time)) (
  PARTITION p2023 VALUES LESS THAN (TO_DAYS('2024-01-01')),
  PARTITION p2024 VALUES LESS THAN (TO_DAYS('2025-01-01')),
  PARTITION p2025 VALUES LESS THAN MAXVALUE
);

-- 数据归档：将2023年数据归档到历史表
CREATE TABLE orders_history LIKE orders;
INSERT INTO orders_history SELECT * FROM orders WHERE pay_time < '2024-01-01';
DELETE FROM orders WHERE pay_time < '2024-01-01';
```

### 3. 优化效果验证

- 索引优化后：查询耗时从**30s+** 降至**1-2s**；
- 分区表优化后：仅扫描 2024 年分区，耗时进一步降至**500ms 内**；
- 归档后：主表数据量减少 30%，聚合效率提升 20%。

### 2. 核心 SQL（递归 + 层级统计）

sql











```sql
-- 步骤1：递归查询组织层级路径
WITH RECURSIVE org_hierarchy AS (
  -- 递归起点：顶级组织
  SELECT
    org_id,
    org_name,
    parent_org_id,
    org_type,
    CAST(org_name AS CHAR(200)) AS org_path
  FROM org WHERE parent_org_id = 0
  
  UNION ALL
  
  -- 递归体：子组织
  SELECT
    o.org_id,
    o.org_name,
    o.parent_org_id,
    o.org_type,
    CONCAT(oh.org_path, '→', o.org_name) AS org_path
  FROM org o
  JOIN org_hierarchy oh ON o.parent_org_id = oh.org_id
),
-- 步骤2：统计每个组织及其所有子组织的用户数
org_user_count AS (
  SELECT
    oh.org_id,
    oh.org_name,
    COUNT(DISTINCT uo.user_id) AS total_user_count
  FROM org_hierarchy oh
  LEFT JOIN org_hierarchy oh_child ON oh_child.org_path LIKE CONCAT(oh.org_path, '%')
  LEFT JOIN user_org uo ON oh_child.org_id = uo.org_id
  GROUP BY oh.org_id, oh.org_name
),
-- 步骤3：统计管理员权限覆盖度
admin_permission AS (
  SELECT
    oh.org_id,
    oh.org_name,
    COUNT(DISTINCT up.permission_code) AS admin_perm_count,
    (SELECT COUNT(DISTINCT permission_code) FROM user_permission) AS total_perm_count,
    ROUND(COUNT(DISTINCT up.permission_code) / (SELECT COUNT(DISTINCT permission_code) FROM user_permission), 4) AS perm_coverage
  FROM org_hierarchy oh
  LEFT JOIN user_org uo ON oh.org_id = uo.org_id AND uo.role = '管理员'
  LEFT JOIN user_permission up ON uo.user_id = up.user_id
  GROUP BY oh.org_id, oh.org_name
),
-- 步骤4：筛选空组织（无子组织+无用户）
empty_org AS (
  SELECT
    oh.org_id,
    oh.org_name
  FROM org_hierarchy oh
  LEFT JOIN org o_child ON oh.org_id = o_child.parent_org_id
  LEFT JOIN user_org uo ON oh.org_id = uo.org_id
  WHERE o_child.org_id IS NULL AND uo.user_id IS NULL
)

-- 最终输出
SELECT
  oh.org_path,
  ouc.total_user_count,
  ap.perm_coverage,
  IF(eo.org_id IS NOT NULL, '是空组织', '非空组织') AS org_status
FROM org_hierarchy oh
LEFT JOIN org_user_count ouc ON oh.org_id = ouc.org_id
LEFT JOIN admin_permission ap ON oh.org_id = ap.org_id
LEFT JOIN empty_org eo ON oh.org_id = eo.org_id;
```

### 3. 关键解析

- **递归查询**：`WITH RECURSIVE` 实现无限层级组织路径拼接；
- **子组织用户统计**：通过`org_path LIKE CONCAT(oh.org_path, '%')` 匹配所有子组织；
- **空组织筛选**：左连接子组织和用户表，筛选 “子组织为空 + 用户为空” 的记录；
- **权限覆盖度**：管理员权限数 / 总权限数，体现组织权限管控水平。

sql











```sql
-- 步骤1：解析配送信息
WITH delivery_stats AS (
  SELECT
    JSON_UNQUOTE(JSON_EXTRACT(ext_json, '$.delivery.province')) AS province,
    COUNT(*) AS order_count,
    AVG(JSON_UNQUOTE(JSON_EXTRACT(ext_json, '$.delivery.fee'))) AS avg_delivery_fee
  FROM order_ext
  GROUP BY province
),
-- 步骤2：解析商品数组（JSON_TABLE拆分多商品）
goods_stats AS (
  SELECT
    g.category,
    SUM(g.price) AS total_amount,
    COUNT(DISTINCT oe.order_id) AS order_count
  FROM order_ext oe
  JOIN JSON_TABLE(
    oe.ext_json->'$.goods',
    '$[*]' COLUMNS (
      id INT PATH '$.id',
      name VARCHAR(50) PATH '$.name',
      category VARCHAR(20) PATH '$.category',
      price DECIMAL(10,2) PATH '$.price'
    )
  ) g
  GROUP BY g.category
),
-- 步骤3：统计新用户+复购标签占比
tag_stats AS (
  SELECT
    COUNT(*) AS total_order,
    COUNT(CASE 
      WHEN JSON_CONTAINS(ext_json->'$.user_tags', '"新用户"') 
      AND JSON_CONTAINS(ext_json->'$.user_tags', '"复购"') 
      THEN 1 END) AS target_order,
    ROUND(COUNT(CASE 
      WHEN JSON_CONTAINS(ext_json->'$.user_tags', '"新用户"') 
      AND JSON_CONTAINS(ext_json->'$.user_tags', '"复购"') 
      THEN 1 END) / COUNT(*), 4) AS target_ratio
  FROM order_ext
),
-- 步骤4：数码品类客单价
digital_avg_price AS (
  SELECT
    SUM(g.price) / COUNT(DISTINCT oe.order_id) AS digital_avg_amount
  FROM order_ext oe
  JOIN JSON_TABLE(
    oe.ext_json->'$.goods',
    '$[*]' COLUMNS (
      category VARCHAR(20) PATH '$.category',
      price DECIMAL(10,2) PATH '$.price'
    )
  ) g
  WHERE g.category = '数码'
)

-- 最终输出
SELECT
  ds.province,
  ds.order_count,
  ds.avg_delivery_fee,
  gs.category,
  gs.total_amount,
  ts.target_ratio,
  dap.digital_avg_amount
FROM delivery_stats ds
CROSS JOIN goods_stats gs
CROSS JOIN tag_stats ts
CROSS JOIN digital_avg_price dap;
```

### 3. 关键解析

- **JSON 数组拆分**：`JSON_TABLE` 是 MySQL 8.0 + 核心功能，将 JSON 数组拆分为行数据，解决多商品统计问题；
- **JSON 筛选**：`JSON_CONTAINS` 判断标签是否存在，`JSON_EXTRACT` 提取嵌套字段；
- **客单价计算**：数码品类总金额 / 数码订单数，需按订单 ID 去重（一个订单含多个数码商品）。

```sql
WITH user_first_order AS (

  GROUP BY ur.user_id, ur.register_time, ur.channel
),
user_rebuy AS (
  SELECT
    ufo.user_id,
    ufo.channel,
    ufo.first_order_time,
    COUNT(o.order_id) AS rebuy_count,
    MIN(o.order_time) AS first_rebuy_time,
    AVG(DATEDIFF(o.order_time, LAG(o.order_time) OVER(PARTITION BY o.user_id ORDER BY o.order_time))) AS avg_rebuy_interval
  FROM user_first_order ufo
  JOIN orders o ON ufo.user_id = o.user_id
  WHERE o.order_time > ufo.first_order_time -- 复购订单
    AND DATEDIFF(o.order_time, ufo.register_time) <= 30 -- 30天内复购
  GROUP BY ufo.user_id, ufo.channel, ufo.first_order_time
),
-- 步骤2：按月统计30日复购率
monthly_rebuy AS (
  SELECT
    DATE_FORMAT(ufo.register_time, '%Y-%m') AS register_month,
    ufo.channel,
    COUNT(DISTINCT ufo.user_id) AS total_new_user,
    COUNT(DISTINCT ur.user_id) AS rebuy_user,
    ROUND(COUNT(DISTINCT ur.user_id) / COUNT(DISTINCT ufo.user_id), 4) AS rebuy_rate
  FROM user_first_order ufo
  LEFT JOIN user_rebuy ur ON ufo.user_id = ur.user_id
  GROUP BY register_month, ufo.channel
),
-- 步骤3：首单/复购用户客单价差异
price_diff AS (
  SELECT
    CASE 
      WHEN o.order_time = ufo.first_order_time THEN '首单用户'
      ELSE '复购用户'
    END AS user_type,
    AVG(o.order_amount) AS avg_amount
  FROM user_first_order ufo
  JOIN orders o ON ufo.user_id = o.user_id
  GROUP BY user_type
)

-- 最终输出：复购率+复购间隔+客单价
SELECT
  mr.register_month,
  mr.channel,
  mr.total_new_user,
  mr.rebuy_user,
  mr.rebuy_rate,
  ur.avg_rebuy_interval,
  pd.user_type,
  pd.avg_amount
FROM monthly_rebuy mr
LEFT JOIN user_rebuy ur ON mr.channel = ur.channel
LEFT JOIN price_diff pd ON 1=1
ORDER BY mr.register_month, mr.rebuy_rate DESC;
```

### 3. 关键解析

- **首单判断**：`MIN(order_time)` 确定用户首单时间，区分首单 / 复购订单；
- **复购间隔**：`LAG()` 窗口函数获取上一次下单时间，`DATEDIFF` 计算间隔；
- **30 日复购率**：按 “注册月份 + 渠道” 分组，统计 30 天内复购用户数 / 新增用户数；
- **客单价差异**：CASE WHEN 区分首单 / 复购用户，计算平均客单价。

## 题目 9：事务 + 触发器实现订单自动分仓

### 场景 & 需求回顾

按收货地址 + 金额自动分配仓库，修改地址 / 金额时自动更新，处理库存异常。

### 1. 建表语句

sql











```sql
-- 订单表（含仓库字段）
CREATE TABLE orders (
  order_id BIGINT PRIMARY KEY AUTO_INCREMENT,
  user_id INT NOT NULL,

  order_amount DECIMAL(10,2) NOT NULL,

  update_time DATETIME NOT NULL DEFAULT NOW()
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 仓库库存表
CREATE TABLE warehouse_stock (
  warehouse VARCHAR(30) PRIMARY KEY,
  stock INT NOT NULL DEFAULT 0 COMMENT '总库存'
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 测试数据
INSERT INTO warehouse_stock VALUES 
('上海仓', 1000), ('杭州仓', 800), ('广州仓', 1200), ('北京仓', 1500);
```

### 2. 核心事务 + 触发器 SQL

#### （1）订单自动分仓事务

sql











```sql
START TRANSACTION;

-- 定义变量
SET @user_id = 1001;
SET @receiver_province = '浙江省';
SET @order_amount = 600.00;

-- 步骤1：计算仓库
SET @warehouse = CASE
  WHEN @receiver_province IN ('上海市', '江苏省', '浙江省', '安徽省') THEN 
    IF(@order_amount >= 500, '杭州仓', '上海仓')
  WHEN @receiver_province IN ('广东省', '广西省', '海南省') THEN '广州仓'
  ELSE '北京仓'
END;

-- 步骤2：检查仓库库存
SELECT stock INTO @stock FROM warehouse_stock WHERE warehouse = @warehouse;
IF @stock > 0 THEN
  -- 库存充足：创建订单
  INSERT INTO orders (user_id, receiver_province, order_amount, warehouse)
  VALUES (@user_id, @receiver_province, @order_amount, @warehouse);
  
  -- 扣减仓库库存
  UPDATE warehouse_stock SET stock = stock - 1 WHERE warehouse = @warehouse;
  
  COMMIT;
  SELECT CONCAT('订单创建成功，分配仓库：', @warehouse) AS result;
ELSE
  -- 库存不足：回滚
  ROLLBACK;
  SELECT CONCAT('仓库', @warehouse, '库存不足，订单创建失败') AS result;
END IF;
```

#### （2）触发器：修改地址 / 金额自动更新仓库

sql











```sql
DELIMITER //
CREATE TRIGGER trg_update_warehouse
BEFORE UPDATE ON orders
FOR EACH ROW
BEGIN
  -- 重新计算仓库
  SET NEW.warehouse = CASE
    WHEN NEW.receiver_province IN ('上海市', '江苏省', '浙江省', '安徽省') THEN 
      IF(NEW.order_amount >= 500, '杭州仓', '上海仓')
    WHEN NEW.receiver_province IN ('广东省', '广西省', '海南省') THEN '广州仓'
    ELSE '北京仓'
  END;
  -- 更新修改时间
  SET NEW.update_time = NOW();
END //
DELIMITER ;

-- 测试触发器
UPDATE orders SET receiver_province = '广东省', order_amount = 400.00 WHERE order_id = 1;
```

#### （3）仓库订单统计

sql











```sql
SELECT
  warehouse,
  COUNT(*) AS order_count,
  SUM(order_amount) AS total_amount,
  ROUND(SUM(order_amount) / (SELECT SUM(order_amount) FROM orders), 4) AS amount_ratio
FROM orders
GROUP BY warehouse;
```

### 3. 关键解析

- **分仓规则落地**：CASE WHEN 实现地址 + 金额的分仓逻辑，可灵活扩展；
- **触发器自动更新**：BEFORE UPDATE 触发器，修改地址 / 金额时自动重新计算仓库；
- **库存异常处理**：事务中先查库存，不足则回滚，保证订单与库存一致性；
- **性能优化**：仓库字段预计算存储，避免每次查询都计算分仓，提升查询效率。

## 题目 10：大数据量用户画像标签计算

### 场景 & 需求回顾

基于 90 天行为生成用户标签，批量更新 + 性能优化 + 交叉标签统计。

### 1. 建表语句

### 2. 核心 SQL（标签计算 + 批量更新）

sql











```sql
-- 步骤1：计算用户标签（基于90天行为）
WITH user_tag_calc AS (
  SELECT
    user_id,
    -- 标签1：数码爱好者（高频浏览数码）
    MAX(CASE 
      WHEN action_type = '浏览' AND action_value = '数码' 
      THEN 1 ELSE 0 END) AS is_digital_fan,
    -- 标签2：高频消费用户（每月下单≥3次）
    MAX(CASE 
      WHEN (SELECT COUNT(*) FROM user_action ua2 
            WHERE ua2.user_id = ua.user_id 
            AND ua2.action_type = '下单' 
            AND ua2.action_time BETWEEN DATE_SUB(ua.action_time, INTERVAL 30 DAY) AND ua.action_time) >= 3 
      THEN 1 ELSE 0 END) AS is_high_freq,
    -- 标签3：高价值用户（收藏金额≥5000）
    MAX(CASE 
      WHEN action_type = '收藏' AND CAST(action_value AS DECIMAL) >= 5000 
      THEN 1 ELSE 0 END) AS is_high_value
  FROM user_action ua
  WHERE action_time BETWEEN DATE_SUB(NOW(), INTERVAL 90 DAY) AND NOW()
  GROUP BY user_id
),
-- 步骤2：拼接标签（去重）
user_tag_merge AS (
  SELECT
    user_id,
    TRIM(BOTH ',' FROM CONCAT(
      CASE WHEN is_digital_fan = 1 THEN '数码爱好者,' ELSE '' END,
      CASE WHEN is_high_freq = 1 THEN '高频消费用户,' ELSE '' END,
      CASE WHEN is_high_value = 1 THEN '高价值用户,' ELSE '' END
    )) AS new_tags
  FROM user_tag_calc
),
-- 步骤3：交叉标签统计
cross_tag_stats AS (
  SELECT
    CONCAT(
      CASE WHEN is_digital_fan = 1 THEN '数码爱好者+' ELSE '' END,
      CASE WHEN is_high_value = 1 THEN '高价值用户' ELSE '' END
    ) AS cross_tag,
    COUNT(user_id) AS user_count
  FROM user_tag_calc
  WHERE is_digital_fan = 1 OR is_high_value = 1
  GROUP BY cross_tag
)

-- 步骤4：批量更新用户标签（新增去重，删除失效标签）
INSERT INTO user_profile (user_id, tags, update_time)
SELECT
  utm.user_id,
  utm.new_tags,
  NOW()
FROM user_tag_merge utm
ON DUPLICATE KEY UPDATE
  tags = utm.new_tags,
  update_time = NOW();

-- 输出交叉标签统计
SELECT * FROM cross_tag_stats;
```

### 3. 性能优化思路

| 优化点       | 具体方案                                                     |
| ------------ | ------------------------------------------------------------ |
| 大数据量过滤 | 分区表按时间分区，仅扫描近 90 天分区；                       |
| 索引优化     | `idx_user_time` 索引加速用户 + 时间筛选；                    |
| 批量更新     | 用`INSERT ... ON DUPLICATE KEY UPDATE` 替代逐条 UPDATE，减少 IO； |
| 并行计算     | 将用户按 ID 分批次，多线程并行计算标签（避免单 SQL 耗时过久）； |
| 标签缓存     | 新增标签临时表，先计算再更新，避免直接更新大表；             |

### 4. 关键解析

- **标签生成**：CASE WHEN 按行为规则生成标签，MAX () 保证用户只要满足一次就标记；
- **标签去重**：CONCAT+TRIM 拼接标签，避免重复逗号；
- **交叉标签统计**：组合不同标签维度，统计用户数，支撑精细化运营；
- **大数据量适配**：分区表 + 批量更新 + 并行计算，保证亿级数据下标签计算效率。

## 总结

### 核心难点 & 解决方案回顾

1. **高并发场景**：乐观锁（版本号）+ 唯一索引，避免超卖 / 重复下单；
2. **复杂统计**：CTE 分层拆分逻辑，窗口函数（LAG/ROW_NUMBER）实现时序分析；
3. **JSON 解析**：JSON_TABLE 拆分数组，JSON_CONTAINS/EXTRACT 处理嵌套字段；
4. **大数据量优化**：分区表 + 联合索引 + 批量操作，减少全表扫描和 IO；
5. **数据一致性**：事务 + 触发器 + 定时校验，保证跨表数据同步；
6. **分库分表**：UNION ALL 合并查询，行号分页避免数据错位。

### 实战落地建议

- 优先使用 CTE 替代多层嵌套子查询，提升 SQL 可读性和维护性；
- 高并发场景优先选乐观锁，低并发 / 数据一致性要求高的场景选悲观锁；
- 大数据量统计优先 “先过滤、后关联、再聚合”，减少中间数据量；
- 业务规则（如分仓、标签）预计算存储，避免查询时实时计算。
