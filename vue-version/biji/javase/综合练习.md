    }

        this.name = name;

2. **StudentScoreSystem.java（成绩管理类：核心逻辑）**

```
import java.util.Scanner;

public class StudentScoreSystem {

    // 存储多个学生对象（数组）
    private Student[] students;
    private int count; // 实际学生数量
    private Scanner sc;

    // 构造方法：初始化数组
    public StudentScoreSystem(int capacity) {
        students = new Student[capacity];
        count = 0;
        sc = new Scanner(System.in);
    }

    // 添加学生
    public void addStudent() {
        if (count >= students.length) {
            System.out.println("数组已满，无法添加！");
            return;
        }

        System.out.print("请输入学号：");
        String id = sc.next();

        System.out.print("请输入姓名：");
        String name = sc.next();

        System.out.print("请输入Java成绩：");
        double java = sc.nextDouble();

        System.out.print("请输入HTML成绩：");
        double html = sc.nextDouble();

        System.out.print("请输入SQL成绩：");
        double sql = sc.nextDouble();

        // 创建学生对象
        Student stu = new Student(id, name, java, html, sql);
        students[count] = stu;
        count++;
        System.out.println("添加成功！");
    }

    // 遍历所有学生
    public void showAll() {
        if (count == 0) {
            System.out.println("暂无学生数据！");
            return;
        }

        System.out.println("==================== 所有学生信息 ====================");
        for (int i = 0; i < count; i++) {
            students[i].showInfo();
        }
        System.out.println("=======================================================");
    }

    // 根据学号查询学生
    public void findById() {
        System.out.print("请输入要查询的学号：");
        String id = sc.next();

        int index = getIndexById(id);
        if (index == -1) {
            System.out.println("该学号不存在！");
        } else {
            students[index].showInfo();
        }
    }

    // 根据学号修改成绩
    public void updateScore() {
        System.out.print("请输入要修改的学生学号：");
        String id = sc.next();

        int index = getIndexById(id);
        if (index == -1) {
            System.out.println("学号不存在！");
            return;
        }

        Student stu = students[index];

        System.out.print("请输入新的Java成绩：");
        double java = sc.nextDouble();
        stu.setJava(java);

        System.out.print("请输入新的HTML成绩：");
        double html = sc.nextDouble();
        stu.setHtml(html);

        System.out.print("请输入新的SQL成绩：");
        double sql = sc.nextDouble();
        stu.setSql(sql);

        System.out.println("修改成功！");
    }

    // 根据学号删除学生
    public void deleteById() {
        System.out.print("请输入要删除的学号：");
        String id = sc.next();

        int index = getIndexById(id);
        if (index == -1) {
            System.out.println("学号不存在！");
            return;
        }

        // 后面元素往前移
        for (int i = index; i < count - 1; i++) {
            students[i] = students[i + 1];
        }
        students[count - 1] = null;
        count--;
        System.out.println("删除成功！");
    }

    // 统计总分最高分
    public void printMaxSumStudent() {
        if (count == 0) {
            System.out.println("暂无学生！");
            return;
        }

        int maxIndex = 0;
        for (int i = 1; i < count; i++) {
            if (students[i].getSum() > students[maxIndex].getSum()) {
                maxIndex = i;
            }
        }
        System.out.println("总分最高的学生是：");
        students[maxIndex].showInfo();
    }

    // 统计各科平均分
    public void printClassAvg() {
        if (count == 0) {
            System.out.println("暂无学生！");
            return;
        }

        double javaSum = 0;
        double htmlSum = 0;
        double sqlSum = 0;

        for (int i = 0; i < count; i++) {
            javaSum += students[i].getJava();
            htmlSum += students[i].getHtml();
            sqlSum += students[i].getSql();
        }

        System.out.println("Java班级平均分：" + (javaSum / count));
        System.out.println("HTML班级平均分：" + (htmlSum / count));
        System.out.println("SQL班级平均分：" + (sqlSum / count));
    }

    // 按总分降序排序（冒泡）
    public void sortBySum() {
        if (count <= 1) return;

        for (int i = 0; i < count - 1; i++) {
            for (int j = 0; j < count - 1 - i; j++) {
                if (students[j].getSum() < students[j + 1].getSum()) {
                    Student temp = students[j];
                    students[j] = students[j + 1];
                    students[j + 1] = temp;
                }
            }
        }
        System.out.println("按总分降序排序完成！");
        showAll();
    }

    // 工具方法：根据学号获取索引
    private int getIndexById(String id) {
        for (int i = 0; i < count; i++) {
            if (students[i].getId().equals(id)) {
                return i;
            }
        }
        return -1;
    }

    // 菜单
    public void menu() {
        while (true) {
            System.out.println("========== 学生成绩管理系统 ==========");
            System.out.println("1. 添加学生");
            System.out.println("2. 查看所有学生");
            System.out.println("3. 根据学号查询");
            System.out.println("4. 修改学生成绩");
            System.out.println("5. 删除学生");
            System.out.println("6. 查看总分最高学生");
            System.out.println("7. 查看班级各科平均分");
            System.out.println("8. 按总分排序");
            System.out.println("0. 退出系统");
            System.out.println("====================================");
            System.out.print("请输入操作：");

            int choice = sc.nextInt();
            switch (choice) {
                case 1:
                    addStudent();
                    break;
                case 2:
                    showAll();
                    break;
                case 3:
                    findById();
                    break;
                case 4:
                    updateScore();
                    break;
                case 5:
                    deleteById();
                    break;
                case 6:
                    printMaxSumStudent();
                    break;
                case 7:
                    printClassAvg();
                    break;
                case 8:
                    sortBySum();
                    break;
                case 0:
                    System.out.println("感谢使用，再见！");
                    return;
                default:
                    System.out.println("输入错误，请重新输入！");
            }
            System.out.println("按回车键继续...");
            sc.nextLine();
            sc.nextLine();
        }
    }
}

/**
 * 图书类：封装图书的属性和行为（Java 对象核心）
 */
public class Book {
    // 1. 私有成员变量（封装核心：private 修饰）
    private int bookId;       // 图书编号
    private String bookName;  // 图书名称
    private String author;    // 作者
    private double price;     // 价格
    private String publish;   // 出版社
    private boolean isBorrow; // 是否被借出：true-已借出 false-未借出

    // 2. 构造方法：创建对象时初始化属性
    // 无参构造
    public Book() {
    }

    // 有参构造
    public Book(int bookId, String bookName, String author, double price, String publish, boolean isBorrow) {
        this.bookId = bookId;
        this.bookName = bookName;
        this.author = author;
        this.price = price;
        this.publish = publish;
        this.isBorrow = isBorrow;
    }

    // 3. getter/setter 方法：外部访问私有属性
    public int getBookId() {
        return bookId;
    }

    public void setBookId(int bookId) {
        if (bookId > 0) {
            this.bookId = bookId;
        } else {
            System.out.println("图书编号不能为负数！");
        }
    }

    public String getBookName() {
        return bookName;
    }

    public void setBookName(String bookName) {
        if (bookName != null && !bookName.trim().isEmpty()) {
            this.bookName = bookName;
        } else {
            System.out.println("图书名称不能为空！");
        }
    }

    public String getAuthor() {
        return author;
    }

    public void setAuthor(String author) {
        this.author = author;
    }

    public double getPrice() {
        return price;
    }

    public void setPrice(double price) {
        if (price >= 0) {
            this.price = price;
        } else {
            System.out.println("图书价格不能为负数！");
        }
    }

    public String getPublish() {
        return publish;
    }

    public void setPublish(String publish) {
        this.publish = publish;
    }

    public boolean isBorrow() {
        return isBorrow;
    }

    public void setBorrow(boolean borrow) {
        isBorrow = borrow;
    }

    // 4. 成员方法：打印图书信息
    public void showBookInfo() {
        System.out.println(bookId + "\t\t" + bookName + "\t\t" + author + "\t\t" + price + "\t\t" + publish + "\t\t" + (isBorrow ? "已借出" : "未借出"));
    }
}

**2.第二个文件：系统主类 `BookManageSystem.java`（核心功能，1000 行）**

```
import java.util.Scanner;

/**
 * 图书管理系统主类
 * 功能：添加、查询、修改、删除、显示所有图书
 */
public class BookManageSystem {
    // 全局常量：最大存储图书数量
    private static final int MAX_BOOKS = 100;
    // 图书数组：存储所有图书对象（核心数据结构）
    private static Book[] bookArray = new Book[MAX_BOOKS];
    // 记录当前图书的实际数量
    private static int bookCount = 0;
    // 控制台输入对象
    private static Scanner scanner = new Scanner(System.in);

    // 静态代码块：初始化默认数据
    static {
        bookArray[0] = new Book(1, "Java编程思想", "布鲁斯·埃克尔", 89.9, "机械工业出版社", false);
        bookArray[1] = new Book(2, "Python从入门到精通", "张三", 59.9, "人民邮电出版社", false);
        bookArray[2] = new Book(3, "MySQL数据库原理", "李四", 49.9, "清华大学出版社", true);
        bookCount = 3;
    }

    // 主方法：程序入口
    public static void main(String[] args) {
        // 循环显示菜单
        while (true) {
            showMainMenu();
            System.out.print("请输入您的操作序号：");
            int choice = scanner.nextInt();
            // 分支控制：根据用户选择执行对应功能
            switch (choice) {
                case 1:
                    addBook();
                    break;
                case 2:
                    queryBook();
                    break;
                case 3:
                    updateBook();
                    break;
                case 4:
                    deleteBook();
                    break;
                case 5:
                    showAllBooks();
                    break;
                case 0:
                    System.out.println("感谢使用图书管理系统，再见！");
                    return;
                default:
                    System.out.println("输入错误，请输入有效数字！");
            }
        }
    }

    /**
     * 显示主菜单
     */
    public static void showMainMenu() {
        System.out.println("==================== 图书管理系统 ====================");
        System.out.println("\t\t1. 添加图书");
        System.out.println("\t\t2. 查询图书");
        System.out.println("\t\t3. 修改图书");
        System.out.println("\t\t4. 删除图书");
        System.out.println("\t\t5. 显示所有图书");
        System.out.println("\t\t0. 退出系统");
        System.out.println("======================================================");
    }

    /**
     * 功能1：添加图书
     */
    public static void addBook() {
        System.out.println("==================== 添加图书 ====================");
        // 判断数组是否已满
        if (bookCount >= MAX_BOOKS) {
            System.out.println("图书库已满，无法添加新图书！");
            return;
        }

        // 接收用户输入
        System.out.print("请输入图书编号：");
        int bookId = scanner.nextInt();
        // 判断编号是否重复
        for (int i = 0; i < bookCount; i++) {
            if (bookArray[i].getBookId() == bookId) {
                System.out.println("该编号已存在，添加失败！");
                return;
            }
        }

        System.out.print("请输入图书名称：");
        String bookName = scanner.next();
        System.out.print("请输入作者：");
        String author = scanner.next();
        System.out.print("请输入价格：");
        double price = scanner.nextDouble();
        System.out.print("请输入出版社：");
        String publish = scanner.next();

        // 创建图书对象
        Book newBook = new Book(bookId, bookName, author, price, publish, false);
        // 将对象存入数组
        bookArray[bookCount] = newBook;
        // 图书数量+1
        bookCount++;
        System.out.println("图书添加成功！");
    }

    /**
     * 功能2：查询图书
     */
    public static void queryBook() {
        System.out.println("==================== 查询图书 ====================");
        System.out.println("1. 按编号查询");
        System.out.println("2. 按名称查询");
        System.out.print("请选择查询方式：");
        int type = scanner.nextInt();

        if (type == 1) {
            System.out.print("请输入图书编号：");
            int id = scanner.nextInt();
            boolean isExist = false;
            // 遍历数组查找
            for (int i = 0; i < bookCount; i++) {
                if (bookArray[i].getBookId() == id) {
                    System.out.println("查询到图书信息：");
                    System.out.println("编号\t\t名称\t\t作者\t\t价格\t\t出版社\t\t状态");
                    bookArray[i].showBookInfo();
                    isExist = true;
                    break;
                }
            }
            if (!isExist) {
                System.out.println("未找到该编号的图书！");
            }
        } else if (type == 2) {
            System.out.print("请输入图书名称：");
            String name = scanner.next();
            boolean isExist = false;
            System.out.println("查询结果：");
            System.out.println("编号\t\t名称\t\t作者\t\t价格\t\t出版社\t\t状态");
            for (int i = 0; i < bookCount; i++) {
                if (bookArray[i].getBookName().contains(name)) {
                    bookArray[i].showBookInfo();
                    isExist = true;
                }
            }
            if (!isExist) {
                System.out.println("未找到该名称的图书！");
            }
        } else {
            System.out.println("输入错误！");
        }
    }

    /**
     * 功能3：修改图书
     */
    public static void updateBook() {
        System.out.println("==================== 修改图书 ====================");
        System.out.print("请输入要修改的图书编号：");
        int id = scanner.nextInt();

        for (int i = 0; i < bookCount; i++) {
            if (bookArray[i].getBookId() == id) {
                System.out.println("找到图书，请输入新的信息：");
                System.out.print("新图书名称：");
                String name = scanner.next();
                System.out.print("新作者：");
                String author = scanner.next();
                System.out.print("新价格：");
                double price = scanner.nextDouble();
                System.out.print("新出版社：");
                String publish = scanner.next();
                System.out.print("是否借出（true/false）：");
                boolean isBorrow = scanner.nextBoolean();

                // 修改对象属性
                bookArray[i].setBookName(name);
                bookArray[i].setAuthor(author);
                bookArray[i].setPrice(price);
                bookArray[i].setPublish(publish);
                bookArray[i].setBorrow(isBorrow);

                System.out.println("图书信息修改成功！");
                return;
            }
        }
        System.out.println("未找到该图书，修改失败！");
    }

    /**
     * 功能4：删除图书
     */
    public static void deleteBook() {
        System.out.println("==================== 删除图书 ====================");
        System.out.print("请输入要删除的图书编号：");
        int id = scanner.nextInt();

        int index = -1;
        // 查找要删除的图书下标
        for (int i = 0; i < bookCount; i++) {
            if (bookArray[i].getBookId() == id) {
                index = i;
                break;
            }
        }

        if (index == -1) {
            System.out.println("未找到该图书，删除失败！");
            return;
        }

        // 数组前移，覆盖删除元素
        for (int i = index; i < bookCount - 1; i++) {
            bookArray[i] = bookArray[i + 1];
        }
        // 最后一位置空
        bookArray[bookCount - 1] = null;
        // 数量减1
        bookCount--;
        System.out.println("图书删除成功！");
    }

    /**
     * 功能5：显示所有图书
     */
    public static void showAllBooks() {
        System.out.println("==================== 所有图书列表 ====================");
        if (bookCount == 0) {
            System.out.println("暂无图书信息！");
            return;
        }
        // 表头
        System.out.println("编号\t\t名称\t\t\t作者\t\t价格\t\t出版社\t\t\t状态");
        System.out.println("------------------------------------------------------");
        // 遍历数组打印所有图书
        for (int i = 0; i < bookCount; i++) {
            bookArray[i].showBookInfo();
        }
        System.out.println("------------------------------------------------------");
        System.out.println("当前图书总数：" + bookCount + " 本");
    }
}
